// 학습 리포트 document.getElementById("reportBtn") .onclick=function(){ const history = JSON.parse( localStorage.getItem("quizHistory") ) || []; if(history.length===0){ alert( "시험 기록이 없습니다." ); return; } let totalScore=0; history.forEach(function(item){ totalScore += item.percent || 0; }); let avg = Math.round( totalScore / history.length ); let weakCategory = JSON.parse( localStorage.getItem("weakCategory") ) || {}; let weak="없음"; let max=0; for(let key in weakCategory){ if(weakCategory[key] > max){ max = weakCategory[key]; weak=key; } } let html = `

나의 학습 리포트

응시 횟수 : ${history.length}회

평균 정답률 : ${avg}%

가장 취약한 분야 : ${weak}

추천 학습 : ${weak} 분야 집중 학습 필요

최근 시험

`; history.slice(-5) .reverse() .forEach(function(item){ html += `

${item.date}
점수 : ${item.score}/${item.total} (${item.percent}%)

`; }); document.querySelector(".container") .innerHTML=html; };
